# Agent Feedback Loop

## Context

AI agents generate code based on rules and templates, but without feedback mechanisms, the same mistakes repeat. This guide establishes a structured approach to:

1. **Track** common errors and anti-patterns generated by AI
2. **Measure** code quality scores over time
3. **Improve** rules based on real-world feedback
4. **Learn** from successful patterns

- [21-code-review.md](./21-code-review.md) - Review rules and severity levels

## 1. Feedback Categories

### 1.1 Error Types

| Category | Description | Example |
|----------|-------------|---------|
| `RULE_VIOLATION` | Generated code violates a known rule | GlobalScope usage |
| `PATTERN_MISMATCH` | Wrong architectural pattern used | Repository in ViewModel |
| `MISSING_ANNOTATION` | Required annotation omitted | Missing @Immutable |
| `WRONG_PARAMETER_ORDER` | Compose params in wrong order | Callbacks before Modifier |
| `INCOMPLETE_OUTPUT` | Code missing required parts | No error handling |
| `INCORRECT_IMPORT` | Wrong or outdated imports | JUnit4 instead of JUnit5 |
| `ANTI_PATTERN` | Known anti-pattern generated | SharedFlow for events |
| `DOCUMENTATION_GAP` | Missing or incorrect docs | No KDoc on public API |

### 1.2 Severity Mapping

```kotlin
enum class FeedbackSeverity {

    WARNING,    // 🟡 Code smell, maintainability issue
    INFO,       // 🔵 Style, optimization suggestion
    SUCCESS,    // ✅ Correct pattern used (positive feedback)
}
```

## 2. Feedback Collection

### 2.1 Feedback Entry Structure

```kotlin
@Immutable
data class AgentFeedback(
    val id: String = UUID.randomUUID().toString(),
    val timestamp: Instant = Instant.now(),
    val category: FeedbackCategory,
    val severity: FeedbackSeverity,
    val ruleId: String,              // e.g., "C1", "W3" from 21-code-review.md
    val ruleName: String,            // e.g., "CancellationException must be rethrown"
    val filePath: String?,
    val lineNumber: Int?,
    val generatedCode: String,       // What AI generated
    val expectedCode: String?,       // What it should be
    val promptUsed: String?,         // The prompt that caused this
    val agentId: String,             // Which AI agent
    val context: Map<String, String> = emptyMap(),
)

enum class FeedbackCategory {
    RULE_VIOLATION,
    PATTERN_MISMATCH,
    MISSING_ANNOTATION,
    WRONG_PARAMETER_ORDER,
    INCOMPLETE_OUTPUT,
    INCORRECT_IMPORT,
    ANTI_PATTERN,
    DOCUMENTATION_GAP,
    SUCCESS,
}
```

### 2.2 Logging Feedback

```kotlin
// In-memory feedback store (for session tracking)
object AgentFeedbackStore {
    private val _feedbacks = mutableListOf<AgentFeedback>()
    val feedbacks: List<AgentFeedback> get() = _feedbacks.toList()
    
    fun log(feedback: AgentFeedback) {
        _feedbacks.add(feedback)
        // Optionally persist to file
        persistToFile(feedback)
    }
    
    fun logViolation(
        ruleId: String,
        ruleName: String,
        generatedCode: String,
        expectedCode: String? = null,
        severity: FeedbackSeverity = FeedbackSeverity.WARNING,
    ) {
        log(AgentFeedback(
            category = FeedbackCategory.RULE_VIOLATION,
            severity = severity,
            ruleId = ruleId,
            ruleName = ruleName,
            generatedCode = generatedCode,
            expectedCode = expectedCode,
            agentId = getCurrentAgentId(),
        ))
    }
    
    fun logSuccess(
        ruleId: String,
        ruleName: String,
        generatedCode: String,
    ) {
        log(AgentFeedback(
            category = FeedbackCategory.SUCCESS,
            severity = FeedbackSeverity.SUCCESS,
            ruleId = ruleId,
            ruleName = ruleName,
            generatedCode = generatedCode,
            agentId = getCurrentAgentId(),
        ))
    }
    
    private fun persistToFile(feedback: AgentFeedback) {
        // Append to feedback log file
        val logLine = buildString {
            append("[${feedback.timestamp}] ")
            append("[${feedback.severity}] ")
            append("[${feedback.ruleId}] ")
            append(feedback.ruleName)
            if (feedback.filePath != null) {
                append(" @ ${feedback.filePath}:${feedback.lineNumber}")
            }
        }
        File("agent-feedback.log").appendText(logLine + "\n")
    }
}
```

## 3. Metrics & Analysis

### 3.1 Key Metrics

```kotlin
data class AgentMetrics(
    val totalGenerations: Int,
    val successfulGenerations: Int,
    val violationCount: Int,
    val criticalViolations: Int,
```

### 3.2 Analysis Functions

```kotlin
object FeedbackAnalyzer {
    
    /**
     * Get top N most violated rules.
     */
    fun getTopViolatedRules(
        feedbacks: List<AgentFeedback>,
        limit: Int = 10,
    ): List<RuleViolationCount> {
        return feedbacks
            .filter { it.category != FeedbackCategory.SUCCESS }
            .groupBy { it.ruleId }
            .map { (ruleId, items) ->
                RuleViolationCount(
                    ruleId = ruleId,
                    ruleName = items.first().ruleName,
                    count = items.size,
                    severity = items.maxOf { it.severity },
                )
            }
            .sortedByDescending { it.count }
            .take(limit)
    }
    
    /**
     * Calculate error rate per category.
     */
    fun getErrorRateByCategory(
        feedbacks: List<AgentFeedback>,
    ): Map<FeedbackCategory, Double> {
        val total = feedbacks.size.toDouble()
        if (total == 0.0) return emptyMap()
        
        return feedbacks
            .groupBy { it.category }
            .mapValues { (_, items) -> items.size / total }
    }
    
    /**
     * Calculate quality score trend over time.
     */
    fun getQualityScoreTrend(
        feedbacks: List<AgentFeedback>,
        days: Int = 30,
    ): List<QualityScore> {
        val endDate = LocalDate.now()
        val startDate = endDate.minusDays(days.toLong())
        
        return feedbacks
            .filter { it.timestamp.toLocalDate() >= startDate }
            .groupBy { it.timestamp.toLocalDate() }
            .map { (date, items) ->
                val violations = items.count { it.category != FeedbackCategory.SUCCESS }
                val score = calculateDailyScore(items.size, violations)
                QualityScore(date, score, items.size)
            }
            .sortedBy { it.date }
    }
    
    private fun calculateDailyScore(total: Int, violations: Int): Int {
        if (total == 0) return 100
        val successRate = (total - violations).toDouble() / total
        return (successRate * 100).toInt()
    }
    
    /**
     * Generate improvement suggestions based on feedback.
     */
    fun generateImprovementSuggestions(
        feedbacks: List<AgentFeedback>,
    ): List<ImprovementSuggestion> {
        val suggestions = mutableListOf<ImprovementSuggestion>()
        val topViolations = getTopViolatedRules(feedbacks, 5)
        
        for (violation in topViolations) {
            when {
                violation.count > 10 && violation.severity == FeedbackSeverity.CRITICAL -> {
                    suggestions.add(ImprovementSuggestion(
                        priority = Priority.HIGH,
                        ruleId = violation.ruleId,
                        suggestion = "Rule '${violation.ruleName}' violated ${violation.count} times. " +
                            "Consider adding more explicit examples to AI_PROMPTS.md",
                        action = "Add negative example to AI_PROMPTS.md section 'Negative Examples'",
                    ))
                }
                violation.count > 5 -> {
                    suggestions.add(ImprovementSuggestion(
                        priority = Priority.MEDIUM,
                        ruleId = violation.ruleId,
                        suggestion = "Rule '${violation.ruleName}' frequently violated. " +
                            "Review rule clarity in guide.",
                        action = "Update rule description in relevant guide",
                    ))
                }
            }
        }
        
        return suggestions.sortedByDescending { it.priority }
    }
}

data class ImprovementSuggestion(
    val priority: Priority,
    val ruleId: String,
    val suggestion: String,
    val action: String,
)

enum class Priority { HIGH, MEDIUM, LOW }
```

```kotlin
/**
 * Rules for auto-generating improvement suggestions.
 */
object AutoSuggestionRules {
    
    /**
     * When a rule is violated more than threshold times,
     * suggest adding it to Negative Examples.
     */
    fun shouldAddNegativeExample(
        ruleId: String,
        violationCount: Int,
        threshold: Int = 5,
    ): Boolean = violationCount >= threshold
    
    /**
     * When error rate for a category exceeds threshold,
     * suggest reviewing related guide.
     */
    fun shouldReviewGuide(
        category: FeedbackCategory,
        errorRate: Double,
        threshold: Double = 0.15,  // 15%
    ): Boolean = errorRate >= threshold
    
    /**
     * When quality score drops below threshold,
     * trigger alert for rule review.
     */
    fun shouldAlertQualityDrop(
        currentScore: Int,
        previousScore: Int,
        dropThreshold: Int = 10,
    ): Boolean = previousScore - currentScore >= dropThreshold
    
    /**
     * Map category to relevant guide for review.
     */
    fun getRelevantGuide(category: FeedbackCategory): String = when (category) {
        FeedbackCategory.RULE_VIOLATION -> "21-code-review.md"
        FeedbackCategory.PATTERN_MISMATCH -> "01-architecture.md"
```

## 5. Integration with Code Review

### 5.1 Auto-Log from @review

When running `@review` or `@review-changes`, automatically log feedback:

```kotlin
/**
 * Integration hook for code review.
 * Call this after each review finding.
 */
fun logReviewFinding(
    finding: ReviewFinding,
    filePath: String,
    lineNumber: Int,
) {
    val severity = when (finding.level) {
        "🔴" -> FeedbackSeverity.CRITICAL
        "🟡" -> FeedbackSeverity.WARNING
        "🔵" -> FeedbackSeverity.INFO
        else -> FeedbackSeverity.INFO
    }
    
    AgentFeedbackStore.log(AgentFeedback(
        category = mapFindingToCategory(finding),
        severity = severity,
        ruleId = finding.ruleId,
        ruleName = finding.ruleName,
        filePath = filePath,
        lineNumber = lineNumber,
        generatedCode = finding.codeSnippet,
        expectedCode = finding.suggestedFix,
        agentId = "code-review",
    ))
}

private fun mapFindingToCategory(finding: ReviewFinding): FeedbackCategory {
    return when {
        finding.ruleId.startsWith("C") -> FeedbackCategory.RULE_VIOLATION
        finding.ruleName.contains("annotation", ignoreCase = true) -> 
            FeedbackCategory.MISSING_ANNOTATION
        finding.ruleName.contains("parameter", ignoreCase = true) -> 
            FeedbackCategory.WRONG_PARAMETER_ORDER
        finding.ruleName.contains("import", ignoreCase = true) -> 
            FeedbackCategory.INCORRECT_IMPORT
        else -> FeedbackCategory.RULE_VIOLATION
    }
}
```
